// before `const needsScene`
/**
 * Scene rules beyond the schema (see contracts/scene-model): repeated siblings must be
 * one procedural node, transformations fine-grained operations, large masses nested.
 * Only repetition has a deterministic repair; the others are accepted after one AI round.
 */
function sceneReview(scene: Scene, among?: Set<string>) {
  const groups = repeatedSiblings(scene, among), issues = granularityIssues(scene, among);
  if (!groups.length && !issues.length) return null;
  return { feedback: [groups.length ? repeatFeedback(groups) : "", issues.length ? granularityFeedback(issues) : ""].filter(Boolean).join("\n\n"),
    repair: () => groups.length ? validateScene(collapseRepeats(scene, groups)) : scene };
}

// in `transform`, after `finalize`
    // A new or regenerated transformation must be one fine-grained operation too.
    review: (input, scene: Scene) => {
      if (input.targetPropertyId) return null;
      const node = findNode(scene, input.targetNodeId)!, placed = findTransformation(node, input.targetTransformationId) || node.transformations.at(-1);
      const issue = placed && transformationIssue(node, placed);
      return issue ? { feedback: `GRANULARITY VIOLATION: ${issue.message}. Return ONE finely grained semantic operation named for what it does (e.g. "Outward size decay", "Wind sway"), with 1–${MAX_FINE_PROPERTIES - 1} properties; never an effect-type bucket.`, repair: () => scene } : null;
    },

// in `revision`, after `finalize`
    // Only new nodes are checked: the user may keep repeated nodes on purpose.
    review: (input, result: SceneProposal) => {
      const review = sceneReview(result.scene, new Set(result.scene.nodes.filter(node => !findNode(input.scene!, node.id)).map(node => node.id)));
      return review && { feedback: review.feedback, repair: () => ({ ...result, scene: review.repair() }) };
    },
